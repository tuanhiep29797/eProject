////////////////////////////////////////////////////
// A. MEMBER
////////////////////////////////////////////////////

Table User {
  user_id int [pk, increment]
  fullname varchar
  username varchar
  email varchar
  phone_number varchar
  password varchar
  role varchar
  created_at datetime
  updated_at datetime
}

////////////////////////////////////////////////////
// B. WEB
////////////////////////////////////////////////////

Table Category {
  category_id int [pk, increment]
  category_name varchar
  category_thumbnail varchar
}

Table Brand {
  brand_id int [pk, increment]
  brand_name varchar
  brand_thumbnail varchar
}

Table Product {
  product_id int [pk, increment]
  product_title varchar
  product_description text
  product_price decimal
  product_content text
  product_quantity int
  product_thumbnail varchar

  product_img_id int
  category_id int
  brand_id int

  created_at datetime
  updated_at datetime
}

Table Product_img {
  product_img_id int [pk, increment]
  product_id int
  url varchar
}

Table Gallery {
  img_id int [pk, increment]
  url varchar
}

////////////////////////////////////////////////////
// CART
////////////////////////////////////////////////////

Table Cart {
  cart_id int [pk, increment]
  user_id int
  product_id int
  quantity int
  created_at datetime
}

////////////////////////////////////////////////////
// REVIEW (đánh giá sản phẩm sau đơn hàng)
////////////////////////////////////////////////////

Table Review {
  review_id int [pk, increment]
  order_id int
  user_id int
  product_id int
  review_content text
  rating int
  is_public boolean
  created_at datetime
}

////////////////////////////////////////////////////
// ORDER
////////////////////////////////////////////////////

Table Order {
  order_id int [pk, increment]
  user_id int
  order_status varchar
  total_amount decimal
  receiver varchar
  address varchar
  order_date datetime
}

Table Order_detail {
  order_id int
  product_id int
  quantity int
  unit_price decimal

  Indexes {
    (order_id, product_id) [pk]
  }
}

////////////////////////////////////////////////////
// FEEDBACK (liên hệ người dùng - form liên hệ)
////////////////////////////////////////////////////

Table Feedback {
  feedback_id int [pk, increment]
  username varchar
  email varchar
  phone_number varchar
  content text
  created_at datetime
}

////////////////////////////////////////////////////
// RELATIONSHIPS
////////////////////////////////////////////////////

Ref: Product.category_id > Category.category_id
Ref: Product.brand_id > Brand.brand_id
Ref: Product.product_img_id > Product_img.product_img_id

Ref: Product_img.product_id > Product.product_id

Ref: Cart.user_id > User.user_id
Ref: Cart.product_id > Product.product_id

Ref: Review.order_id > Order.order_id
Ref: Review.user_id > User.user_id
Ref: Review.product_id > Product.product_id

Ref: Order.user_id > User.user_id

Ref: Order_detail.order_id > Order.order_id
Ref: Order_detail.product_id > Product.product_id
